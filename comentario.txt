# Configuración de MySQL 5.7 para Binlog Live

## Archivo: /etc/mysql/mysql.conf.d/mysqld.cnf

[mysqld]
log_bin = mysql-bin
server_id = 1
binlog_format = ROW
binlog_row_image = FULL
expire_logs_days = 7
max_binlog_size = 100M
bind-address = 0.0.0.0

## Permisos del usuario monitor:
GRANT REPLICATION SLAVE, REPLICATION CLIENT, SELECT ON *.* TO 'monitor'@'%';
FLUSH PRIVILEGES;

## Reiniciar MySQL:
sudo systemctl restart mysql
